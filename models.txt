from extensions import db
from sqlalchemy_serializer import SerializerMixin


class SchoolClass(db.Model, SerializerMixin):
    __tablename__ = "classes"

    id = db.Column(db.Integer, primary_key=True)
    

    class_name = db.Column(
        db.String(100),
        nullable=False,
        unique=True
    )

    description = db.Column(db.Text)

    resources = db.relationship(
        "Resource",
        back_populates="school_class",
        cascade="all, delete-orphan"

    )

    class_subjects = db.relationship(
        "ClassSubject",
        back_populates="school_class",
        cascade="all, delete-orphan"
    )
    student_classes = db.relationship(
    "StudentClass",
    back_populates="school_class",
    cascade="all, delete-orphan"
    )

    # Add the teacher_classes relationship here
    teacher_classes = db.relationship(
        "TeacherClass",
        back_populates="school_class",
        cascade="all, delete-orphan"
    )


    serialize_rules = (
    '-teacher_classes.school_class',
    '-resources.school_class',
    '-class_subjects.school_class',
    '-student_classes.school_class'
    )

    def __repr__(self):
        return f"<SchoolClass {self.class_name}>"

from extensions import db
from sqlalchemy_serializer import SerializerMixin
from datetime import datetime, timezone


class Resource(db.Model, SerializerMixin):
    __tablename__ = "resources"

    id = db.Column(db.Integer, primary_key=True)

    title = db.Column(
        db.String(255),
        nullable=False
    )

    description = db.Column(db.Text)

    file_name = db.Column(
        db.String(255),
        nullable=False
    )

    file_path = db.Column(
        db.Text,
        nullable=False
    )

    file_size = db.Column(db.BigInteger)

    file_type = db.Column(
        db.String(50)
    )

    resource_type = db.Column(
        db.Enum(
            'NOTE',
            'REVISION_QUESTION',
            'PAST_PAPER',
            'ASSIGNMENT',
            name='resource_types'
        )
    )

    status = db.Column(
        db.Enum(
            'ACTIVE',
            'ARCHIVED',
            'DELETED',
            name='resource_status'
        ),
        default='ACTIVE'
    )

    subject_id = db.Column(
        db.Integer,
        db.ForeignKey('subjects.id'),
        nullable=False
    )

    class_id = db.Column(
        db.Integer,
        db.ForeignKey('classes.id'),
        nullable=False
    )

    uploaded_by = db.Column(
        db.Integer,
        db.ForeignKey('users.id'),
        nullable=False
    )

    created_at = db.Column(
        db.DateTime,
        default=lambda: datetime.now(timezone.utc)
    )

    updated_at = db.Column(
        db.DateTime,
        default=lambda: datetime.now(timezone.utc),
        onupdate=lambda: datetime.now(timezone.utc)
    )

    subject = db.relationship(
        "Subject",
        back_populates="resources"
    )

    school_class = db.relationship(
        "SchoolClass",
        back_populates="resources"
    )
    

    uploader = db.relationship(
        "User",
        back_populates="uploaded_resources"
    )
    

    downloads = db.relationship(
        "Download",
        back_populates="resource",
        cascade="all, delete-orphan"
    )
    

    resource_tags = db.relationship(
        "ResourceTag",
        back_populates="resource",
        cascade="all, delete-orphan"
    )

    resource_views = db.relationship(
    "ResourceView",
    back_populates="resource",
    cascade="all, delete-orphan"
)
    

    serialize_rules = (
    '-subject.resources',
    '-school_class.resources',
    '-uploader.uploaded_resources',
    '-downloads.resource',
    '-resource_tags.resource',
    '-resource_views.resource'
    )

    def __repr__(self):
        return f"<Resource {self.title}>"

from extensions import db, bcrypt
from sqlalchemy_serializer import SerializerMixin
from sqlalchemy.ext.hybrid import hybrid_property
from datetime import datetime, timezone
from sqlalchemy.orm import validates
import re


class User(db.Model, SerializerMixin):
    __tablename__ = "users"

    id = db.Column(db.Integer, primary_key=True)

    first_name = db.Column(
        db.String(100),
        nullable=False
    )

    last_name = db.Column(
        db.String(100),
        nullable=False
    )

    email = db.Column(
        db.String(255),
        unique=True,
        nullable=False
    )

    _password_hash = db.Column(
        db.String(255),
        nullable=False
    )

    role_id = db.Column(
        db.Integer,
        db.ForeignKey('roles.id'),
        nullable=False
    )

    is_active = db.Column(
        db.Boolean,
        default=True
    )

    last_login = db.Column(db.DateTime)

    created_at = db.Column(
        db.DateTime,
        default=lambda: datetime.now(timezone.utc)
    )

    updated_at = db.Column(
        db.DateTime,
        default=lambda: datetime.now(timezone.utc),
        onupdate=lambda: datetime.now(timezone.utc)
    )

    role = db.relationship(
        "Role",
        back_populates="users"
    )

    uploaded_resources = db.relationship(
        "Resource",
        back_populates="uploader"
    )
    student_classes = db.relationship(
    "StudentClass",
    back_populates="student",
    cascade="all, delete-orphan"
    )

    downloads = db.relationship(
        "Download",
        back_populates="user",
        cascade="all, delete-orphan"
    )
    resource_views = db.relationship(
    "ResourceView",
    back_populates="user",
    cascade="all, delete-orphan"
)
    # Add the teacher_classes relationship here
    teacher_classes = db.relationship(
        "TeacherClass",
        back_populates="teacher",
        cascade="all, delete-orphan"
    )



    serialize_rules = (
    '-teacher_classes.teacher',
    '-_password_hash',
    '-role.users',
    '-student_classes.student',
    '-downloads.user',
    '-resource_views.user',
    '-uploaded_resources.uploader'
    )

    #password handling property

    @hybrid_property
    def password(self):
        raise AttributeError("Password is not readable")
    


    @password.setter
    def password(self, plain_password):
        self._password_hash =bcrypt.generate_password_hash(plain_password).decode('utf-8')


    def check_password(self, plain_password):
        return bcrypt.check_password_hash(self._password_hash, plain_password)

    @validates('email')
    def validate_email(self,key,email):
        email = email.lower()
        if not re.match(r"[^@]+@[^@]+\.[^@]+", email):
            raise ValueError('invalid email format')
        return email
    

    

    def __repr__(self):
        return f"<User {self.email}>"

from extensions import db

    teacher_id = db.Column(
        db.Integer,
        db.ForeignKey('users.id'),
        nullable=False
    )

    class_id = db.Column(
        db.Integer,
        db.ForeignKey('classes.id'),
        nullable=False
    )

    # Relationships
    teacher = db.relationship(
        "User", 
        back_populates="teacher_classes"
    )
    
    school_class = db.relationship(
        "SchoolClass", 
        back_populates="teacher_classes"
    )

    # Ensures a teacher can't be added to the exact same class twice

        db.UniqueConstraint('teacher_id', 'class_id', name='uq_teacher_class'),
    )

    serialize_rules = (
        '-teacher.teacher_classes',
        '-school_class.teacher_classes',
    )

    def __repr__(self):
