from flask import make_response, request

# Setup a production logger to track errors silently without leaking them to users
logger = logging.getLogger(__name__)

def resource_getall_service():
    try:
        # Senior standard: Modern, explicit SQLAlchemy 2.0 select statement
        resources = db.session.scalars(select(Resource)).all()
        return make_response(resources_schema.dump(resources), 200)
    except SQLAlchemyError as e:
        logger.error(f"Database error in resource_getall_service: {str(e)}")
        return {"error": "An unexpected error occurred while retrieving resources."}, 500

def resource_post():
    # Senior Guardrail: Gracefully handle missing or malformed JSON payloads
    data = request.get_json(silent=True)
    if data is None:
        return {"error": "Invalid JSON or missing request body."}, 400

    try:
        validated_data = resource_schema.load(data)
    except ValidationError as e:
        return {"error": e.messages}, 400
    
    # Senior Note: If file_path is omitted in the schema for security, 
    # the service layer MUST securely generate it since the model requires it (nullable=False).
    # Replace this placeholder with your actual secure cloud/file storage generation path logic.
    secure_generated_path = f"uploads/resources/{validated_data.get('file_name', 'default.dat')}"

    new_resource = Resource(
        title=validated_data.get('title'),
        description=validated_data.get('description'),
        file_name=validated_data.get('file_name'),
        file_path=secure_generated_path,  # Satisfies the model's non-nullable constraint safely
        file_size=validated_data.get('file_size'),
        file_type=validated_data.get('file_type'),
        resource_type=validated_data.get('resource_type'),
        status=validated_data.get('status', 'ACTIVE'),
        subject_id=validated_data.get('subject_id'),
        class_id=validated_data.get('class_id'),
        uploaded_by=validated_data.get('uploaded_by')
    )

    try:
        db.session.add(new_resource)
        db.session.commit()
        return make_response(resource_schema.dump(new_resource), 201)
    except SQLAlchemyError as e:
        db.session.rollback()  # Crucial: Reset the session state so subsequent requests don't break
        logger.error(f"Database insertion failed: {str(e)}")
        return {"error": "Database error occurred while creating the resource."}, 500

def resource_getbyid_services(id):
    try:
        resource = db.session.get(Resource, id)  
        if not resource:
            return {"message": "Resource not found"}, 404
        return make_response(resource_schema.dump(resource), 200)
    except SQLAlchemyError as e:
        logger.error(f"Database error in resource_getbyid_services for ID {id}: {str(e)}")
        return {"error": "An unexpected error occurred."}, 500

def resource_pathbyid_service(id):
    try:
        resource = db.session.get(Resource, id)
        if not resource:
            return {"message": "Resource not found"}, 404
    except SQLAlchemyError as e:
        logger.error(f"Database read failure during patch for ID {id}: {str(e)}")
        return {"error": "Database lookup failed."}, 500
    
    patch_data = request.get_json(silent=True)
    if patch_data is None:
        return {"error": "Invalid JSON payload provided."}, 400

    try:
        validated_data = resource_schema.load(patch_data, partial=True)
    except ValidationError as e:
        return {"error": e.messages}, 400
    
    # Safely apply properties to the fetched model instance
    for key, value in validated_data.items():
        if hasattr(resource, key):
            setattr(resource, key, value)
            
    try:
        db.session.commit()
        return make_response(resource_schema.dump(resource), 200)
    except SQLAlchemyError as e:
        db.session.rollback()
        logger.error(f"Database patch transaction update failed for ID {id}: {str(e)}")
        return {"error": "Could not finalize resource updates."}, 500

def resource_deletebyid_service(id):
    try:
        resource = db.session.get(Resource, id)
        if not resource:
            return {"message": "Resource not found"}, 404
            
        db.session.delete(resource)
        db.session.commit()
        return make_response({"message": "Resource deleted successfully"}, 200)
    except SQLAlchemyError as e:
        db.session.rollback()
        logger.error(f"Database deletion transaction failed for ID {id}: {str(e)}")
        return {"error": "Failed to delete resource due to a database conflict."}, 500

def dashboard_search_service(query, page, per_page):
    try:        
        if not query:
            return {
                "status": "success",
                "data": {
                    "resource": [],
                    "users" : [],
                    "classes": []
                }
            }, 200
        
        offset = (page-1)*per_page

        # Resources Search
        resources = db.session.execute(
            select(Resource.id.label('resource_id'), 
                   Resource.title.label('resource_title')
            )
            .where(Resource.title.ilike(f"%{query}%"))
            .limit(per_page)
            .offset(offset)
        ).all()

        # Users Search
        users = db.session.execute(
            select(User.id.label('user_id'), 
                   User.username.label('user_name'),
                   User.email.label('user_email')
            )
            .where(
                or_(
                    User.username.ilike(f"%{query}%"),
                    User.email.ilike(f"%{query}%")
                )
            )
            .limit(per_page)
            .offset(offset)
        ).all()

        # Classes Search
        classes = db.session.execute(
            select(SchoolClass.id.label('class_id'), 
                   SchoolClass.class_name.label('class_classname')
            )
            .where(SchoolClass.class_name.ilike(f"%{query}%"))
            .limit(per_page)
            .offset(offset)
        ).all()

        payload = {
            'resources': resources, 
            'users': users,
            'classes': classes 
        }

        # Fixed: now correctly calls .dump() on the non-many instance
        return {
            "status": "success",
            "page":page,
            "per_page":per_page,
            "data" :dashboard_Search_schema.dump(payload)
        }, 200

    except SQLAlchemyError as e:
        logger.error(f"Search error: {str(e)}")
        return {"error": "Search failed"}, 500
