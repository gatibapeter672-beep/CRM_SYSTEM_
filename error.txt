import axios from 'axios';

const apiClient = axios.create({
  baseURL: '/api',
  withCredentials: true, //🚀 CRITICAL: Forces browser to include cookies automatically
  headers: {
    'Content-Type': 'application/json'
  },
});

// 🔄 AUTOMATIC TOKEN INTERCEPTOR RESCUE MATRIX
// Inside your apiClient.js response interceptor:
apiClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    const originalRequest = error.config;
    
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;
      try {
        // Post request automatically passes the hidden refresh token cookie
        await axios.post('/api/refresh', {}, { withCredentials: true });
        return apiClient(originalRequest); // Retry original endpoint request
      } catch (refreshError) {
        // 🚀 IF REFRESH FAILS: Broadcast an event to let AuthContext know the session died
        window.dispatchEvent(new Event('auth:session_expired'));
        return Promise.reject(refreshError);
      }
    }
    return Promise.reject(error);
  }
);

import React, { createContext, useState, useEffect, useContext, useCallback } from 'react';
import apiClient from './apiClient';

export const AuthContext = createContext(null);

// Helper function to normalize user role to a clean, lowercase string
const normalizeUserRole = (userData) => {
  if (!userData) return null;
  
  // If role is an object like { role_name: 'admin' }, extract the string
  const rawRole = typeof userData.role === 'object' 
    ? userData.role?.role_name 
    : userData.role;
    
  return {
    ...userData,
    role: rawRole ? rawRole.toLowerCase().trim() : null
  };
};

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  const logout = useCallback(async () => {
    try {
      await apiClient.post('/logout');
    } catch (err) {
      console.error("Session logout cleanup error:", err);
    } finally {
      setUser(null);
      localStorage.removeItem('user');
      localStorage.removeItem('user_role');
    }
  }, []);

  // Initial Boot: Verify the cookie session with the backend
  useEffect(() => {
    const initializeAuth = async () => {
      const savedUser = localStorage.getItem('user');
      
      if (savedUser) {
        try {
          setUser(JSON.parse(savedUser));
          
          // Fast backend ping to ensure session is valid
          const response = await apiClient.get('/me'); 
          console.log("ME RESPONSE:", response.data);
          // 🚀 FIX: Normalize the backend user data before setting state
          const normalized = normalizeUserRole(response.data.user);
          console.log("NORMALIZED USER:", normalized);
          setUser(normalized);
          localStorage.setItem('user', JSON.stringify(normalized));
          if (normalized?.role) {
            localStorage.setItem('user_role', normalized.role);
          }
        } catch (err) {
          console.error("Session verification failed. Clearing state.");
          setUser(null);
          localStorage.removeItem('user');
          localStorage.removeItem('user_role');
        }
      }
      setLoading(false);
    };

    initializeAuth();
  }, []);

  useEffect(() => {
    const handleForceLogout = () => {
      setUser(null);
      localStorage.removeItem('user');
      localStorage.removeItem('user_role');
    };

    window.addEventListener('auth:session_expired', handleForceLogout);
    return () => window.removeEventListener('auth:session_expired', handleForceLogout);
  }, []);

  const login = (userData) => {
    // 🚀 FIX: Normalize here as well to keep everything aligned
    const normalized = normalizeUserRole(userData);
    setUser(normalized);
    localStorage.setItem('user', JSON.stringify(normalized));
    if (normalized?.role) {
      localStorage.setItem('user_role', normalized.role);
    }
  };

  const isAuthenticated = !!user;

  return (
    <AuthContext.Provider
      value={{
        user,
        isAuthenticated,
        login,
        logout,
        loading,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);

from functools import wraps
from flask_jwt_extended import verify_jwt_in_request, get_jwt_identity
from models import User

def role_required(*allowed_roles):
    def wrapper(fn):
        @wraps(fn)
        def decorator(*args, **kwargs):
            # ensure JWT is valid
            verify_jwt_in_request()

            user_id = get_jwt_identity()
            user = User.query.get(user_id)

            if not user:
                return {"message": "User not found"}, 404

            if not user.is_active:
                return {"message": "User account is inactive"}, 403

            user_role = user.role.role_name.lower()

            if user_role not in [role.lower() for role in allowed_roles]:
                return {"message": "Access Forbidden: insufficient permissions"}, 403

            return fn(*args, **kwargs)

        return decorator
    return wrapper
