from flask_jwt_extended import jwt_required, get_jwt, get_jwt_identity, create_access_token
from flask_restful import Resource
from models.token_blocklist import TokenBlocklist
from extensions import db
from app import api
from datetime import datetime, timezone

class Logout(Resource):
    @jwt_required(verify_type=False) # Allows both access and refresh tokens to hit this endpoint
    def post(self):
        jwt_payload = get_jwt()
        jti = jwt_payload["jti"]
        token_type = jwt_payload["type"]
        
        # Convert JWT exp timestamp to a timezone-aware datetime object
        expires_at = datetime.fromtimestamp(jwt_payload["exp"], tz=timezone.utc)

        # Check if already blacklisted to avoid unique constraint crashes
        exists = TokenBlocklist.query.filter_by(jti=jti).first()
        if not exists:
            blocked_token = TokenBlocklist(
                jti=jti,
                token_type=token_type,
                expires_at=expires_at
            )
            db.session.add(blocked_token)
            db.session.commit()

        return {"message": f"{token_type.capitalize()} token revoked successfully"}, 200


class Refresh(Resource):
    @jwt_required(refresh=True)
    def post(self):
        user_id = get_jwt_identity()
        new_access_token = create_access_token(identity=user_id)

        return {"access_token": new_access_token}, 200
    
api.add_resource(Logout, '/logout')

api.add_resource(Refresh, '/refresh')

from models import User, Role
from app import db
from schemas.registrationSchema import registration_schema
from marshmallow import ValidationError


def register_user_service(data):

    if not data:
        return {"error": "No input data provided"}, 400

    # validate
    try:
        validated_data = registration_schema.load(data)
    except ValidationError as e:
        return {"error": e.messages}, 400

    # check email
    if User.query.filter_by(email=validated_data.email).first():
        return {"error": "Email already registered"}, 400

    # check role
    role = db.session.get(Role, validated_data.role_id)
    if not role:
        return {"error": "Role not found"}, 404

    # create user
    new_user = User(
        first_name=validated_data.first_name,
        last_name=validated_data.last_name,
        email=validated_data.email.lower(),
        # password=validated_data.password,    password should not e read according models
        role_id=validated_data.role_id
        
    )
    # 🔐 SET password (DO NOT READ IT) 
    new_user.password = data["password"]

    db.session.add(new_user)
    db.session.commit()

    return {
        "message": "User registered successfully",
        "user": registration_schema.dump(new_user)
    }, 201

# server/services/decoration.py

from functools import wraps
from flask_jwt_extended import verify_jwt_in_request, get_jwt_identity
from models import User

def role_required(*allowed_roles):
    def wrapper(fn):
        @wraps(fn)
        def decorator(*args, **kwargs):
            # ensure JWT is valid
            verify_jwt_in_request()

            user_id = get_jwt_identity()
            user = User.query.get(user_id)

            if not user:
                return {"message": "User not found"}, 404

            if not user.is_active:
                return {"message": "User account is inactive"}, 403

            user_role = user.role.role_name.lower()

            if user_role not in [role.lower() for role in allowed_roles]:
                return {"message": "Access Forbidden: insufficient permissions"}, 403

            return fn(*args, **kwargs)

        return decorator
    return wrapper

# server/services/auth_service.py

from models import User,Role
from app import db
from schemas.loginSchema import login_schema
from marshmallow import ValidationError
from schemas.registrationSchema import registration_schema


from datetime import datetime, timezone, timedelta
from flask_jwt_extended import create_access_token, create_refresh_token
from flask import make_response


def login_user_service(data):
    if not data:
        return {"message": "no input data provided"}, 400

    try:
        validated_data = login_schema.load(data)
    except ValidationError as e:
        return {"errors": e.messages}, 400

    email = validated_data.email.lower()
    raw_password = data["password"]

    user = User.query.filter_by(email=email).first()

    if not user or not user.check_password(raw_password) or not user.is_active:
        return {"message": "invalid credentials"}, 401

    user.last_login = datetime.now(timezone.utc)
    db.session.commit()

    access_token = create_access_token(identity=user.id, expires_delta=timedelta(hours=1))
    refresh_token = create_refresh_token(identity=user.id)

    return make_response({
        "access_token": access_token,
        "refresh_token": refresh_token,
        "user": registration_schema.dump(user)
    }, 200)

import React, { createContext, useState, useContext,useEffect } from 'react';

export const AuthContext = createContext();

const USER_KEY = 'user';
const TOKEN_KEY = 'token';

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [token, setToken] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const storedUser = localStorage.getItem(USER_KEY);
    const storedToken = localStorage.getItem(TOKEN_KEY);
    if (storedUser && storedToken) {
      setUser(JSON.parse(storedUser));
      setToken(storedToken);
    }
    setLoading(false);
  }, []);

  const login = (userData, authToken) => {
    setUser(userData);
    setToken(authToken);
    localStorage.setItem(USER_KEY, JSON.stringify(userData));
    localStorage.setItem(TOKEN_KEY, authToken);
  };

  const logout = () => {
    setUser(null);
    setToken(null);
    localStorage.removeItem(USER_KEY);
    localStorage.removeItem(TOKEN_KEY);
  };

  const refreshAuth = () => {
    const storedUser = localStorage.getItem(USER_KEY);
    const storedToken = localStorage.getItem(TOKEN_KEY);
    if (storedUser && storedToken) {
      setUser(JSON.parse(storedUser));
      setToken(storedToken);
    }
  };

  const isAuthenticated = !!token;

  return (
    <AuthContext.Provider
      value={{
        user,
        token,
        isAuthenticated,
        login,
        logout,
        refreshAuth,
        loading,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);

// src/layouts/ProtectedRoute.jsx
import React from 'react';
import { Navigate, Outlet } from 'react-router-dom';
import useAuth from '../hooks/useAuth'; // Assumes your hooks folder lives under src/

export default function ProtectedRoute({ allowedRoles = [] }) {
  const { isAuthenticated, user, loading } = useAuth();

  // 1. Prevent premature redirection while checking auth tokens
  if (loading) {
    return (
      <div className="min-h-screen bg-slate-50 flex flex-col items-center justify-center text-sm font-semibold text-slate-500 gap-2">
        <div className="h-5 w-5 rounded-full border-2 border-emerald-600 border-t-transparent animate-spin" />
        <span>Authenticating session parameters...</span>
      </div>
    );
  }

  // 2. Bounce back to login if unauthorized
  if (!isAuthenticated || !user) {
    return <Navigate to="/login" replace />;
  }

  // 3. Send to unauthorized deck if the role doesn't match
  if (allowedRoles.length > 0 && !allowedRoles.includes(user.role)) {
    return <Navigate to="/unauthorized" replace />;
  }

  // 4. Upgraded: Pass user data down via Outlet Context!
  // This allows any inner page to access the logged-in user instantly.
  return <Outlet context={{ user }} />;
}
